function initModuleCards() {
    const cards = document.querySelectorAll('.module-card');
    cards.forEach(card => {
        card.addEventListener('touchstart', () => { card.style.transform = 'scale(0.98)'; });
        card.addEventListener('touchend', () => { card.style.transform = ''; });
    });
}
